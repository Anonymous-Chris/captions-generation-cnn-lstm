%%writefile /content/test_polos.py

from polos.models import download_model, load_checkpoint
from PIL import Image

model_path = download_model("polos")
model = load_checkpoint(model_path)

data = [
    {
        "img": Image.open("/content/10815824_2997e03d76.jpg").convert("RGB"),
        "mt": "a man is standing next to a dog",
        "refs": [
            "a man standing beside a dog",
            "a person is standing next to a dog",
            "a man and a dog are outside"
        ],
    }
]

_, scores = model.predict(
    data,
    batch_size=1,
    cuda=True
)

print("POLOS score:", scores)

print("POLOS score:", scores)
